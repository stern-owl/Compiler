# LL(1) Grammar for the Language
# Compiler Project - CS F363
#
# Format: non-terminal -> symbol1 symbol2 ... | eps
# Terminals start with TK_
# eps represents epsilon production
#
# Modified to match the professor's Modified LL(1) Grammar document
# (February 13, 2026)

# 1. Start Symbol
program -> otherFunctions mainFunction

# 2. Main Function
mainFunction -> TK_MAIN stmts TK_END

# 3. Other Functions
otherFunctions -> function otherFunctions
otherFunctions -> eps

# 4. Function Definition
function -> TK_FUNID input_par output_par TK_SEM stmts TK_END

# 5. Input Parameters
input_par -> TK_INPUT TK_PARAMETER TK_LIST TK_SQL parameter_list TK_SQR

# 6. Output Parameters (can be empty)
output_par -> TK_OUTPUT TK_PARAMETER TK_LIST TK_SQL parameter_list TK_SQR
output_par -> eps

# 7. Parameter List
parameter_list -> dataType TK_ID remaining_list

# 8. Data Types
dataType -> primitiveDatatype
dataType -> constructedDatatype

# 9. Primitive Data Types
primitiveDatatype -> TK_INT
primitiveDatatype -> TK_REAL

# 10-a. Constructed Data Types (includes TK_RUID for alias/definetype)
constructedDatatype -> TK_RECORD TK_RUID
constructedDatatype -> TK_UNION TK_RUID
constructedDatatype -> TK_RUID

# 11. Remaining parameters
remaining_list -> TK_COMMA parameter_list
remaining_list -> eps

# 12. Statements Block
stmts -> typeDefinitions declarations otherStmts returnStmt

# 13-a. Type Definitions (using actualOrRedefined grouping)
typeDefinitions -> actualOrRedefined typeDefinitions
typeDefinitions -> eps

# 13-b. Actual or Redefined
actualOrRedefined -> typeDefinition
actualOrRedefined -> definetypestmt

# 14, 15. Type Definition (Record/Union)
typeDefinition -> TK_RECORD TK_RUID fieldDefinitions TK_ENDRECORD
typeDefinition -> TK_UNION TK_RUID fieldDefinitions TK_ENDUNION

# 16. Field Definitions (at least 2 fields)
fieldDefinitions -> fieldDefinition fieldDefinition moreFields

# 17. Single Field Definition (supports nested via fieldType)
fieldDefinition -> TK_TYPE fieldType TK_COLON TK_FIELDID TK_SEM

# 17-a. Field Type (primitive or constructed - alias only for nesting)
fieldType -> primitiveDatatype
fieldType -> constructedDatatype

# 18. More Fields (optional)
moreFields -> fieldDefinition moreFields
moreFields -> eps

# 19. Declarations
declarations -> declaration declarations
declarations -> eps

# 20. Single Declaration
declaration -> TK_TYPE dataType TK_COLON TK_ID global_or_not TK_SEM

# 21. Global or Not (modified: colon is part of global_or_not)
global_or_not -> TK_COLON TK_GLOBAL
global_or_not -> eps

# 22. Other Statements
otherStmts -> stmt otherStmts
otherStmts -> eps

# 23. Statement Types
stmt -> assignmentStmt
stmt -> iterativeStmt
stmt -> conditionalStmt
stmt -> ioStmt
stmt -> funCallStmt

# 24. Assignment Statement
assignmentStmt -> singleOrRecId TK_ASSIGNOP arithmeticExpression TK_SEM

# 25 e. Single or Record ID (left factored, supports nested expansion)
singleOrRecId -> TK_ID option_single_constructed

# 25 f. Option: plain ID or constructed variable
option_single_constructed -> oneExpansion moreExpansions
option_single_constructed -> eps

# 25 c. One level of field expansion
oneExpansion -> TK_DOT TK_FIELDID

# 25 d. More expansions (recursive for nested records)
moreExpansions -> oneExpansion moreExpansions
moreExpansions -> eps

# 26. Function Call Statement
funCallStmt -> outputParameters TK_CALL TK_FUNID TK_WITH TK_PARAMETERS inputParameters TK_SEM

# 27. Output Parameters (optional)
outputParameters -> TK_SQL idList TK_SQR TK_ASSIGNOP
outputParameters -> eps

# 28. Input Parameters
inputParameters -> TK_SQL idList TK_SQR

# 29. Iterative Statement (while loop)
iterativeStmt -> TK_WHILE TK_OP booleanExpression TK_CL stmt otherStmts TK_ENDWHILE

# 30-a. Conditional Statement (with TK_OP/TK_CL around boolean expression)
conditionalStmt -> TK_IF TK_OP booleanExpression TK_CL TK_THEN stmt otherStmts elsePart

# 31-a. Else Part (else requires at least one stmt)
elsePart -> TK_ELSE stmt otherStmts TK_ENDIF
elsePart -> TK_ENDIF

# 32. I/O Statements
ioStmt -> TK_READ TK_OP var TK_CL TK_SEM
ioStmt -> TK_WRITE TK_OP var TK_CL TK_SEM

# B1. Arithmetic Expression (left recursion removed, precedence added)
arithmeticExpression -> term expPrime

# B2. Expression Prime
expPrime -> lowPrecedenceOperators term expPrime
expPrime -> eps

# B3. Term
term -> factor termPrime

# B4. Term Prime
termPrime -> highPrecedenceOperators factor termPrime
termPrime -> eps

# B5. Factor
factor -> TK_OP arithmeticExpression TK_CL
factor -> var

# B6. High Precedence Operators
highPrecedenceOperators -> TK_MUL
highPrecedenceOperators -> TK_DIV

# B7. Low Precedence Operators
lowPrecedenceOperators -> TK_PLUS
lowPrecedenceOperators -> TK_MINUS

# 36. Boolean Expression (logical)
booleanExpression -> TK_OP booleanExpression TK_CL logicalOp TK_OP booleanExpression TK_CL

# 37. Boolean Expression (relational)
booleanExpression -> var relationalOp var

# 38-a. Boolean Expression (NOT - with parentheses)
booleanExpression -> TK_NOT TK_OP booleanExpression TK_CL

# 39-a. Variable (uses singleOrRecId for nested field access)
var -> singleOrRecId
var -> TK_NUM
var -> TK_RNUM

# 40. Logical Operators
logicalOp -> TK_AND
logicalOp -> TK_OR

# 41. Relational Operators
relationalOp -> TK_LT
relationalOp -> TK_LE
relationalOp -> TK_EQ
relationalOp -> TK_GT
relationalOp -> TK_GE
relationalOp -> TK_NE

# 42. Return Statement
returnStmt -> TK_RETURN optionalReturn TK_SEM

# 43. Optional Return Values
optionalReturn -> TK_SQL idList TK_SQR
optionalReturn -> eps

# 44. ID List
idList -> TK_ID more_ids

# 45. More IDs
more_ids -> TK_COMMA idList
more_ids -> eps

# 46. Define Type Statement
definetypestmt -> TK_DEFINETYPE typeConstructor TK_RUID TK_AS TK_RUID

# 47. Type Constructor (A)
typeConstructor -> TK_RECORD
typeConstructor -> TK_UNION
